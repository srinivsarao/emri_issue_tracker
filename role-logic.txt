<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, ...$roles)
    {
        if (! auth()->check()) {
            return redirect('/login');
        }

        if (! auth()->user()->hasAnyRole($roles)) {
            abort(403);
        }

        return $next($request);
    }
}

    public function handle($request, Closure $next, ...$roles)
    {
        if (!auth()->check()) {
            return redirect('/login');

        if (!in_array(auth()->user()->role_id, $roles)) {
            abort(403);
        }

        return $next($request);
    }
}

namespace App\Http\Controllers;

use App\Constants\Role;

class DashboardController extends Controller
{
    public function index()
    {
        $user = auth()->user();

        switch ($user->role_id) {

            case Role::CENTRAL_ADMIN:
                return redirect()->route('central.dashboard');

            case Role::STATE_ADMIN:
                return redirect()->route('state.dashboard');

            case Role::STATE_IT:
                return redirect()->route('state.it.dashboard');

            case Role::HO_ADMIN:
                return redirect()->route('ho.dashboard');

            case Role::HO_IT_MANAGER:
                return redirect()->route('ho.manager.dashboard');

            case Role::HO_IT_ENGINEER:
                return redirect()->route('ho.engineer.dashboard');

            case Role::VENDOR_ADMIN:

            case Role::VENDOR_MANAGER:
                return redirect()->route('vendor.manager.dashboard');

            case Role::VENDOR_ENGINEER:
                return redirect()->route('vendor.engineer.dashboard');

            case Role::REPORT_VIEWER:
                return redirect()->route('report.dashboard');

            default:
                abort(403,'Access Denied');
        }
    }
}

<?php

namespace App\Http\Controllers\CentralAdmin;

use App\Http\Controllers\Controller;
use App\Models\State;
use Illuminate\Http\Request;
use Carbon\Carbon;


class StateController extends Controller
{

    public function index()
    {
        $states = State::orderBy('state_name')
                ->paginate(10);

        return view(
            'central.states.index',
            compact('states')
        );
    }



    public function create()
    {
        return view('central.states.create');
    }




    public function store(Request $request)
    {

        $request->validate([

            'state_code'=>'required|max:10|unique:mst_state,state_code',

            'state_name'=>'required|max:100',

        ]);



        State::create([

            'state_code'=>$request->state_code,

            'state_name'=>$request->state_name,

            'description'=>$request->description,

            'is_active'=>1,

            'created_by'=>auth()->id(),

            'created_date'=>Carbon::now()

        ]);



        return redirect()
        ->route('central.states.index')
        ->with('success','State created successfully');

    }





    public function edit(State $state)
    {
        return view(
            'central.states.edit',
            compact('state')
        );
    }





    public function update(Request $request, State $state)
    {


        $request->validate([

            'state_code'
            =>
            'required|max:10|unique:mst_state,state_code,'.$state->state_id.',state_id',


            'state_name'
            =>
            'required|max:100'

        ]);



        $state->update([

            'state_code'=>$request->state_code,

            'state_name'=>$request->state_name,

            'description'=>$request->description,

            'updated_by'=>auth()->id(),

            'updated_date'=>Carbon::now()

        ]);



        return redirect()
        ->route('central.states.index')
        ->with('success','State updated successfully');

    }





    public function destroy(State $state)
    {

        $state->update([

            'is_active'=>0

        ]);


        return redirect()
        ->route('central.states.index')
        ->with('success','State deactivated');

    }

}

<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class CentralAdmin
{
    /**
     * Handle an incoming request.
     */
    public function handle(Request $request, Closure $next): Response
    {

        // Check login
        if (!auth()->check()) {

            return redirect()
                ->route('login');
        }


        // Central Admin Role ID = 1
        if (auth()->user()->role_id != 1) {

            abort(403, 'Access denied. Central Admin only.');

        }


        return $next($request);
    }
}
